cmake_minimum_required(VERSION 3.20)
project(VueVPNNative LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0")
set(CMAKE_OSX_ARCHITECTURES "arm64")
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/openvpn3")
find_package(OpenSSL REQUIRED)
find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
find_path(ASIO_INCLUDE_DIR asio.hpp REQUIRED)
find_path(LZ4_INCLUDE_DIR lz4.h REQUIRED)
add_library(vuevpn-core STATIC "${CORE}/client/ovpncli.cpp" "${CORE}/openvpn/crypto/data_epoch.cpp")
target_include_directories(vuevpn-core PUBLIC "${CORE}" "${ASIO_INCLUDE_DIR}" "${LZ4_INCLUDE_DIR}")
# Each attempt owns its reactor. Once Core finishes graceful shutdown, return
# from connect() even if an unrelated async operation still holds work alive.
target_compile_definitions(vuevpn-core PUBLIC USE_TUN_BUILDER ASIO_STANDALONE USE_ASIO HAVE_LZ4 USE_OPENSSL OPENVPN_IO_REQUIRES_STOP)
target_link_libraries(vuevpn-core PUBLIC OpenSSL::SSL OpenSSL::Crypto "${LZ4_LIBRARY}" "-framework SystemConfiguration" "-framework IOKit" "-framework CoreFoundation")
add_executable(vuevpn-helper src/Helper.mm src/Engine.mm src/Network.mm src/Connectivity.mm)
target_include_directories(vuevpn-helper PRIVATE include)
target_compile_options(vuevpn-helper PRIVATE -fobjc-arc)
target_link_libraries(vuevpn-helper PRIVATE vuevpn-core "-framework Foundation" "-framework Security" "-framework Network")
enable_testing()
add_executable(native-unit-tests tests/PolicyTests.cpp)
target_include_directories(native-unit-tests PRIVATE include)
target_compile_options(native-unit-tests PRIVATE -UNDEBUG)
add_test(NAME native-policy COMMAND native-unit-tests)
add_executable(core-unit-tests tests/CoreTests.cpp)
target_include_directories(core-unit-tests PRIVATE include)
target_compile_options(core-unit-tests PRIVATE -UNDEBUG)
target_link_libraries(core-unit-tests PRIVATE vuevpn-core)
add_test(NAME embedded-core-parser COMMAND core-unit-tests)

# In-memory lifecycle tests: never links the real Network implementation.
add_executable(engine-unit-tests tests/EngineTests.mm)
target_include_directories(engine-unit-tests PRIVATE include)
target_compile_options(engine-unit-tests PRIVATE -UNDEBUG -fobjc-arc)
target_link_libraries(engine-unit-tests PRIVATE vuevpn-core "-framework Foundation")
add_test(NAME engine-lifecycle COMMAND engine-unit-tests)
set_tests_properties(engine-lifecycle PROPERTIES TIMEOUT 30)

# Real Network logic with fake OS boundaries; no privileged operations.
add_executable(network-unit-tests tests/NetworkTests.mm)
target_include_directories(network-unit-tests PRIVATE include)
target_compile_options(network-unit-tests PRIVATE -UNDEBUG -fobjc-arc)
target_link_libraries(network-unit-tests PRIVATE "-framework Foundation" "-framework SystemConfiguration")
add_test(NAME network-ownership COMMAND network-unit-tests)
set_tests_properties(network-ownership PROPERTIES TIMEOUT 15)
